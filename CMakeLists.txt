cmake_minimum_required(VERSION 3.20)
project(OPMChord VERSION 1.2.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
elseif(MINGW)
  add_link_options(-municode)
endif()
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE src)
enable_testing()
add_test(NAME core COMMAND core_tests)
if(WIN32)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    add_executable(OPMChord WIN32 src/main.cpp)
    target_include_directories(OPMChord PRIVATE src)
    target_compile_definitions(OPMChord PRIVATE UNICODE _UNICODE)
    target_link_libraries(OPMChord PRIVATE winmm user32 gdi32 comctl32)
    add_custom_command(TARGET OPMChord POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/bin/independent_opm.dll" "$<TARGET_FILE_DIR:OPMChord>/independent_opm.dll")
  else()
    add_executable(OPMChord-X68Bridge32 src/x68_helper.cpp)
    target_include_directories(OPMChord-X68Bridge32 PRIVATE src)
    target_compile_definitions(OPMChord-X68Bridge32 PRIVATE UNICODE _UNICODE)
  endif()
endif()
